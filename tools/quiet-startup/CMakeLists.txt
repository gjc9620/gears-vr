cmake_minimum_required(VERSION 3.20)
project(GearsQuietStartup LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_subdirectory(../third-party/minhook minhook)
add_library(GearsQuietStartup SHARED quiet_startup.cpp)
target_link_libraries(GearsQuietStartup PRIVATE minhook user32 shell32 ole32 uuid)
add_executable(InstancePolicyTest instance_policy_test.cpp)
target_link_libraries(InstancePolicyTest PRIVATE shell32 ole32 uuid)
target_compile_definitions(InstancePolicyTest PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(InstancePolicyTest PRIVATE /W4 /EHsc /UNDEBUG)
add_executable(GearsEarlyLaunch early_launch.cpp)
target_link_libraries(GearsEarlyLaunch PRIVATE user32)
add_executable(QuietStartupFixture WIN32 fixture.cpp)
target_link_libraries(QuietStartupFixture PRIVATE user32 shell32)
add_executable(QuietCommandLineTest command_line_test.cpp)
add_executable(BackgroundPolicyTest test_background_policy_test.cpp)
target_compile_options(BackgroundPolicyTest PRIVATE /UNDEBUG)
target_link_libraries(QuietCommandLineTest PRIVATE shell32)
enable_testing()
add_test(NAME QuietCommandLineTest COMMAND QuietCommandLineTest)
add_test(NAME BackgroundPolicyTest COMMAND BackgroundPolicyTest)
add_test(NAME InstancePolicyTest COMMAND InstancePolicyTest $<TARGET_FILE:GearsQuietStartup> ${CMAKE_CURRENT_BINARY_DIR}/instance-policy-fixture)
foreach(target GearsQuietStartup GearsEarlyLaunch QuietStartupFixture QuietCommandLineTest BackgroundPolicyTest)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(${target} PRIVATE /W4 /EHsc)
endforeach()
