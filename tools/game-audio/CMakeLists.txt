cmake_minimum_required(VERSION 3.20)
project(GearsTestAudio LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The audio guard requires Windows Core Audio")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(GearsTestAudioGuard game_audio_guard.cpp)
target_link_libraries(GearsTestAudioGuard PRIVATE ole32 uuid)
add_executable(AudioGuardPolicyTest audio_guard_policy_test.cpp)
target_compile_options(AudioGuardPolicyTest PRIVATE /UNDEBUG)
foreach(target GearsTestAudioGuard AudioGuardPolicyTest)
  target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
endforeach()
enable_testing()
add_test(NAME AudioGuardPolicyTest COMMAND AudioGuardPolicyTest)
