cmake_minimum_required(VERSION 3.20)
project(GearsPlayground LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_subdirectory(../third-party/minhook minhook)
set(MH_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../third-party/minhook")
file(READ "${MH_SOURCE_DIR}/src/buffer.c" MH_BUFFER)
string(ASCII 239 187 191 MH_UTF8_BOM)
string(REGEX REPLACE "^${MH_UTF8_BOM}" "" MH_BUFFER "${MH_BUFFER}")
string(FIND "${MH_BUFFER}" "#define MAX_MEMORY_RANGE 0x40000000" MH_RANGE_MATCH)
if(MH_RANGE_MATCH EQUAL -1)
    message(FATAL_ERROR "Playground MinHook range patch no longer matches the reviewed source")
endif()
# This bridge attests a six-byte push-only ProcessEvent prologue. Its trampoline
# has no RIP-relative operand; use the rel32 window with a 64 KiB safety margin.
# Keep these changes local to this target; the renderer's MinHook stays frozen.
string(REPLACE "#define MAX_MEMORY_RANGE 0x40000000" "#define MAX_MEMORY_RANGE 0x7fff0000" MH_BUFFER "${MH_BUFFER}")
string(PREPEND MH_BUFFER "#include <windows.h>\nstatic volatile LONG gears_alloc_attempts,gears_alloc_error;\nDWORD GearsPlaygroundAllocationAttempts(void){return (DWORD)gears_alloc_attempts;}\nDWORD GearsPlaygroundAllocationError(void){return (DWORD)gears_alloc_error;}\nstatic LPVOID GearsPlaygroundVirtualAlloc(LPVOID p,SIZE_T n,DWORD kind,DWORD protect){\n InterlockedIncrement(&gears_alloc_attempts);\n LPVOID result=VirtualAlloc(p,n,kind,protect);\n if(!result)InterlockedExchange(&gears_alloc_error,(LONG)GetLastError());\n return result;\n}\n#define VirtualAlloc GearsPlaygroundVirtualAlloc\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/gears_minhook_buffer.c" "${MH_BUFFER}")
file(READ "${MH_SOURCE_DIR}/src/hook.c" MH_HOOK)
foreach(MH_ALLOCATION_SITE "LPVOID pBuffer = AllocateBuffer(pTarget);" "PHOOK_ENTRY pHook = AddHookEntry();")
    string(FIND "${MH_HOOK}" "${MH_ALLOCATION_SITE}" MH_SITE_MATCH)
    if(MH_SITE_MATCH EQUAL -1)
        message(FATAL_ERROR "Playground MinHook allocation diagnostic no longer matches the reviewed source")
    endif()
endforeach()
string(REPLACE "LPVOID pBuffer = AllocateBuffer(pTarget);"
    "LPVOID pBuffer = AllocateBuffer(pTarget);\n                SetLastError(pBuffer ? ERROR_SUCCESS : ERROR_NOT_ENOUGH_MEMORY);" MH_HOOK "${MH_HOOK}")
string(REPLACE "PHOOK_ENTRY pHook = AddHookEntry();"
    "PHOOK_ENTRY pHook = AddHookEntry();\n                        SetLastError(pHook ? ERROR_SUCCESS : ERROR_OUTOFMEMORY);" MH_HOOK "${MH_HOOK}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/gears_minhook_hook.c" "${MH_HOOK}")
get_target_property(MH_SOURCES minhook SOURCES)
set(MH_REPLACED_SOURCES)
foreach(MH_SOURCE IN LISTS MH_SOURCES)
    if(NOT MH_SOURCE MATCHES "(^|/)(buffer|hook)\\.c$")
        if(IS_ABSOLUTE "${MH_SOURCE}")
            list(APPEND MH_REPLACED_SOURCES "${MH_SOURCE}")
        else()
            list(APPEND MH_REPLACED_SOURCES "${MH_SOURCE_DIR}/${MH_SOURCE}")
        endif()
    endif()
endforeach()
list(APPEND MH_REPLACED_SOURCES "${CMAKE_CURRENT_BINARY_DIR}/gears_minhook_buffer.c")
list(APPEND MH_REPLACED_SOURCES "${CMAKE_CURRENT_BINARY_DIR}/gears_minhook_hook.c")
set_property(TARGET minhook PROPERTY SOURCES "${MH_REPLACED_SOURCES}")
target_include_directories(minhook PRIVATE "${MH_SOURCE_DIR}/src")
add_library(GearsPlayground SHARED playground.cpp)
set(GEARS_PLAYGROUND_OUTPUT_NAME "GearsPlayground" CACHE STRING "Private candidate DLL name")
set_target_properties(GearsPlayground PROPERTIES OUTPUT_NAME "${GEARS_PLAYGROUND_OUTPUT_NAME}")
target_include_directories(GearsPlayground PRIVATE ../third-party/minhook/include)
target_link_libraries(GearsPlayground PRIVATE minhook)
target_compile_definitions(GearsPlayground PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)

add_executable(PlaygroundHookTest playground_hook_test.cpp)
target_link_libraries(PlaygroundHookTest PRIVATE minhook)
target_include_directories(PlaygroundHookTest PRIVATE ../third-party/minhook/include)
target_compile_definitions(PlaygroundHookTest PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
