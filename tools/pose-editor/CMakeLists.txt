cmake_minimum_required(VERSION 3.20)
project(GearsPoseEditorChecks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
if(NOT MSVC)
    message(FATAL_ERROR "The runtime and original-rig fixtures require Windows/MSVC")
endif()
foreach(target IN ITEMS PoseEditorTest PoseEditorRuntimeTest PoseEditorC0Test)
    if(target STREQUAL "PoseEditorTest")
        set(source pose_editor_test.cpp)
    elseif(target STREQUAL "PoseEditorRuntimeTest")
        set(source pose_editor_runtime_test.cpp)
    else()
        set(source pose_editor_c0_test.cpp)
    endif()
    add_executable(${target} ${source})
    target_compile_options(${target} PRIVATE /W4 /WX /EHsc /UNDEBUG)
endforeach()
target_link_libraries(PoseEditorC0Test PRIVATE bcrypt)
