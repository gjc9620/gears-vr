cmake_minimum_required(VERSION 3.20)
project(GearsQuestWorkbench LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(ANDROID)
    if(NOT OPENXR_AAR)
        message(FATAL_ERROR "OPENXR_AAR must name the extracted pinned Khronos loader AAR")
    endif()
    add_library(native_app_glue STATIC "${ANDROID_NDK}/sources/android/native_app_glue/android_native_app_glue.c")
    target_include_directories(native_app_glue PUBLIC "${ANDROID_NDK}/sources/android/native_app_glue")
    add_library(openxr_loader SHARED IMPORTED)
    set_target_properties(openxr_loader PROPERTIES IMPORTED_LOCATION "${OPENXR_AAR}/jni/arm64-v8a/libopenxr_loader.so")
    add_library(quest_workbench SHARED native.cpp render.cpp feedback.cpp)
    target_include_directories(quest_workbench PRIVATE "${OPENXR_AAR}/prefab/modules/headers/include")
    target_compile_definitions(quest_workbench PRIVATE XR_USE_PLATFORM_ANDROID XR_USE_GRAPHICS_API_OPENGL_ES)
    target_compile_options(quest_workbench PRIVATE -Wall -Wextra -Wno-missing-field-initializers -fexceptions -frtti)
    target_link_options(quest_workbench PRIVATE "-Wl,-u,ANativeActivity_onCreate" "-Wl,-z,max-page-size=16384")
    target_link_libraries(quest_workbench PRIVATE native_app_glue openxr_loader android log EGL GLESv3 aaudio)
else()
    add_executable(WorkbenchCoreTest core_test.cpp)
    if(MSVC)
        target_compile_options(WorkbenchCoreTest PRIVATE /W4 /EHsc /utf-8 /UNDEBUG)
    endif()
endif()

if(NOT ANDROID)
    add_executable(WorkbenchPracticeTest practice_test.cpp)
    add_executable(WorkbenchArmoryTest armory_test.cpp)
    if(MSVC)
        target_compile_options(WorkbenchPracticeTest PRIVATE /W4 /EHsc /utf-8 /UNDEBUG)
        target_compile_options(WorkbenchArmoryTest PRIVATE /W4 /EHsc /utf-8 /UNDEBUG)
    endif()
endif()
