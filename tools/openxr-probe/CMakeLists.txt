cmake_minimum_required(VERSION 3.20)
project(GearsOpenXRProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(XR_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../third-party/openxr-loader")
add_executable(OpenXRProbe probe.cpp)
target_include_directories(OpenXRProbe PRIVATE "${XR_ROOT}/include")
target_link_libraries(OpenXRProbe PRIVATE "${XR_ROOT}/x64/lib/openxr_loader.lib")
target_compile_definitions(OpenXRProbe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(OpenXRProbe PRIVATE /W4 /EHsc)
add_custom_command(TARGET OpenXRProbe POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${XR_ROOT}/x64/bin/openxr_loader.dll" "$<TARGET_FILE_DIR:OpenXRProbe>")
add_executable(OpenXRPoseProbe pose_probe.cpp)
target_include_directories(OpenXRPoseProbe PRIVATE "${XR_ROOT}/include")
target_link_libraries(OpenXRPoseProbe PRIVATE "${XR_ROOT}/x64/lib/openxr_loader.lib" d3d12 dxgi)
target_compile_definitions(OpenXRPoseProbe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX XR_USE_PLATFORM_WIN32 XR_USE_GRAPHICS_API_D3D12)
target_compile_options(OpenXRPoseProbe PRIVATE /W4 /EHsc)
add_custom_command(TARGET OpenXRPoseProbe POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${XR_ROOT}/x64/bin/openxr_loader.dll" "$<TARGET_FILE_DIR:OpenXRPoseProbe>")
set_property(TARGET OpenXRPoseProbe PROPERTY CXX_STANDARD 20)
target_link_libraries(OpenXRPoseProbe PRIVATE d3d11 dwmapi windowsapp ole32)
target_link_libraries(OpenXRPoseProbe PRIVATE d3dcompiler)
