# This build has heavily fragmented low address space; the nearest aligned free
# page can sit just beyond MinHook's default 1 GiB search. 1.5 GiB retains margin
# within rel32's 2 GiB range. Generate a build-local copy, keeping vendor pristine.
set(MH_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../third-party/minhook")
file(READ "${MH_SOURCE_DIR}/src/buffer.c" MH_BUFFER)
string(FIND "${MH_BUFFER}" "#define MAX_MEMORY_RANGE 0x40000000" MH_RANGE_POSITION)
if(MH_RANGE_POSITION LESS 0)
    message(FATAL_ERROR "Unexpected MinHook source; allocator patch rejected")
endif()
string(REPLACE "#define MAX_MEMORY_RANGE 0x40000000" "#define MAX_MEMORY_RANGE 0x60000000" MH_BUFFER "${MH_BUFFER}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/gears_minhook_buffer.c" "${MH_BUFFER}")
get_target_property(MH_SOURCES minhook SOURCES)
set(MH_REPLACED_SOURCES)
foreach(MH_SOURCE IN LISTS MH_SOURCES)
    if(NOT MH_SOURCE MATCHES "(^|/)buffer\\.c$")
        if(IS_ABSOLUTE "${MH_SOURCE}")
            list(APPEND MH_REPLACED_SOURCES "${MH_SOURCE}")
        else()
            list(APPEND MH_REPLACED_SOURCES "${MH_SOURCE_DIR}/${MH_SOURCE}")
        endif()
    endif()
endforeach()
list(APPEND MH_REPLACED_SOURCES "${CMAKE_CURRENT_BINARY_DIR}/gears_minhook_buffer.c")
set_property(TARGET minhook PROPERTY SOURCES "${MH_REPLACED_SOURCES}")
target_include_directories(minhook PRIVATE "${MH_SOURCE_DIR}/src")
