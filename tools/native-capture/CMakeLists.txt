cmake_minimum_required(VERSION 3.20)
project(GearsNativeCapture LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_subdirectory(../third-party/minhook minhook)
# This build has heavily fragmented low address space; the nearest aligned free
# page can sit just beyond MinHook's default 1 GiB search. 1.5 GiB retains margin
# within rel32's 2 GiB range. Generate a build-local copy, keeping vendor pristine.
set(MH_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../third-party/minhook")
file(READ "${MH_SOURCE_DIR}/src/buffer.c" MH_BUFFER)
string(FIND "${MH_BUFFER}" "#define MAX_MEMORY_RANGE 0x40000000" MH_RANGE_POSITION)
if(MH_RANGE_POSITION LESS 0)
    message(FATAL_ERROR "Unexpected MinHook source; allocator patch rejected")
endif()
string(REPLACE "#define MAX_MEMORY_RANGE 0x40000000" "#define MAX_MEMORY_RANGE 0x60000000" MH_BUFFER "${MH_BUFFER}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/gears_minhook_buffer.c" "${MH_BUFFER}")
get_target_property(MH_SOURCES minhook SOURCES)
set(MH_REPLACED_SOURCES)
foreach(MH_SOURCE IN LISTS MH_SOURCES)
    if(NOT MH_SOURCE MATCHES "(^|/)buffer\\.c$")
        if(IS_ABSOLUTE "${MH_SOURCE}")
            list(APPEND MH_REPLACED_SOURCES "${MH_SOURCE}")
        else()
            list(APPEND MH_REPLACED_SOURCES "${MH_SOURCE_DIR}/${MH_SOURCE}")
        endif()
    endif()
endforeach()
list(APPEND MH_REPLACED_SOURCES "${CMAKE_CURRENT_BINARY_DIR}/gears_minhook_buffer.c")
set_property(TARGET minhook PROPERTY SOURCES "${MH_REPLACED_SOURCES}")
target_include_directories(minhook PRIVATE "${MH_SOURCE_DIR}/src")
add_library(GearsCapture SHARED capture.cpp)
if(DEFINED CAPTURE_OUTPUT_NAME)
    set_target_properties(GearsCapture PROPERTIES OUTPUT_NAME "${CAPTURE_OUTPUT_NAME}")
endif()
target_link_libraries(GearsCapture PRIVATE minhook d3d12 dxgi bcrypt)
target_compile_definitions(GearsCapture PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(GearsCapture PRIVATE /W4 /EHsc)
add_executable(CaptureSelfTest self_test.cpp)
add_executable(SourceViewShadowTest source_shadow_test.cpp)
target_compile_definitions(SourceViewShadowTest PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(SourceViewShadowTest PRIVATE /W4 /EHsc)
add_executable(SourceViewArgsTest source_args_test.cpp)
add_executable(SourceEyeArgsTest source_eye_args_test.cpp)
add_executable(TwoViewRendererShadowTest two_view_renderer_shadow_test.cpp)
add_executable(SplitSourceViewsTest split_source_views_test.cpp)
add_executable(SubmittedViewLifetimeTest submitted_view_lifetime_test.cpp)
add_executable(GpuViewportStatsTest gpu_viewport_stats_test.cpp)
target_compile_definitions(SplitSourceViewsTest PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(SplitSourceViewsTest PRIVATE /W4 /EHsc)
target_compile_definitions(TwoViewRendererShadowTest PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(TwoViewRendererShadowTest PRIVATE /W4 /EHsc)
target_compile_definitions(SourceEyeArgsTest PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(SourceEyeArgsTest PRIVATE /W4 /EHsc)
target_compile_definitions(SourceViewArgsTest PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
