powershell -NoProfile -ExecutionPolicy Bypass -File tools/vr-launcher/ui/build.ps1 -OutputDirectory analysis/release-rebuild/gui -NoInstall
& $buildPython -B tools/vr-launcher/stage_portable.py --python-root (Split-Path $buildPython) --destination analysis/release-staging/GearsVR-0.0.1-rebuild --manifest analysis/release-0.0.1/frozen/manifest.json --launcher analysis/release-rebuild/gui/GearsVrLauncher.exe --release-version 0.0.1 --preflight-output analysis/release-rebuild/preflight.json
& $buildPython -B tools/vr-launcher/package_portable.py --stage analysis/release-staging/GearsVR-0.0.1-rebuild --output analysis/release-rebuild/GearsVR-0.0.1.zip
